@startuml

class Funcionario {
  + nome: String
  + cpf: String
  + data_nascimento: Date
  + telefone: String
  + cargo: String
  + salario: double
  + horas_semanais: double
}

class Mecanico {
  + especialidade: String
}

class ConsultorTecnico {
}

' Herança
Funcionario <|-- Mecanico
Funcionario <|-- ConsultorTecnico

class Cliente {
  + nome: String
  + cpf: String
  + data_nascimento: Date
  + telefone: String
}

class Carro {
  + vin: String
  + placa: String
  + modelo: String
  + ano: int
  + quilometragem: int
  + cliente: Cliente
}

Cliente "1" -- "0..n" Carro

class OrdemServico {
  + estado: String
  + data_inicio: DateTime
  + previsao_termino: DateTime
  + data_termino: DateTime
  + descricao: String
  + observacao: String
  + carro: Carro
  + mecanicos: List<Mecanicos>
  + consultor_tecnico: ConsultorTecnico
  + total: Decimal
}

' Associações da Ordem de Serviço
Carro "1" -- "0..n" OrdemServico
Mecanico "1..n" -- "0..n" OrdemServico
ConsultorTecnico "1" -- "0..n" OrdemServico

class OrdemPecas {
  + ordem_servico: OrdemServico
  + peca: Peca
  + quantidade: int
  + preco_unitario: Decimal
  + subtotal: Decimal
}

class Peca {
  + nome: String
  + fabricante: String
  + codigo: String
  + descricao: String
  + preco: double
  + estoque: int
}

OrdemServico "1" -- "0..n" OrdemPecas
OrdemPecas "0..n" -- "1" Peca

class Servico {
  + nome: String
  + descricao: String
  + preco: Decimal
}

class OrdemServicoServico {
  + ordem_servico: OrdemServico
  + servico: Servico
  + preco_unitario: Decimal
}

OrdemServico "1" -- "0..n" OrdemServicoServico
OrdemServicoServico "0..n" -- "1" Servico

note right of OrdemServico
  total = soma dos precos dos servicos
  + soma(preco_unitario da peca * quantidade)
end note

@enduml
